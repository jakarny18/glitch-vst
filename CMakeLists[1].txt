cmake_minimum_required(VERSION 3.22) # Link the MSVC runtime statically (/MT) so the exe and the VST3 do not need the # Visual C++ Redistributable installed on the target PC. set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>") project(YouGlitchSMP1024 VERSION 0.1.0) # Keep normal Release optimisation while emitting symbols for crash diagnosis. if (MSVC) add_compile_options(/Zi) add_link_options(/DEBUG) endif() # JUCE is fetched at configure time. Requires an internet connection on first configure. set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE) set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE) include(FetchContent) FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE) FetchContent_MakeAvailable(JUCE) # Diagnostic isolation: prevent JUCE's automatic image-format lookup from # selecting its PNG decoder. The application source does not load PNG assets. # This keeps the other JUCE image formats and all audio functionality intact. set(_juce_image_formats_cpp "${juce_SOURCE_DIR}/modules/juce_graphics/images/juce_ImageFileFormat.cpp") if (NOT EXISTS "${_juce_image_formats_cpp}") message(FATAL_ERROR "JUCE image-format source was not found: ${_juce_image_formats_cpp}") endif() file(READ "${_juce_image_formats_cpp}" _juce_image_formats_source) set(_juce_png_registration [=[ formats[0] = &png; formats[1] = &jpg; formats[2] = &gif; formats[3] = nullptr;]=]) set(_juce_no_png_registration [=[ formats[0] = &jpg; formats[1] = &gif; formats[2] = nullptr;]=]) string(FIND "${_juce_image_formats_source}" "${_juce_png_registration}" _juce_png_registration_pos) if (_juce_png_registration_pos EQUAL -1) message(FATAL_ERROR "Expected JUCE 8.0.6 default PNG registration was not found; refusing an unverified patch.") endif() string(REPLACE "${_juce_png_registration}" "${_juce_no_png_registration}" _juce_image_formats_source "${_juce_image_formats_source}") string(REPLACE "PNGImageFormat png;" "" _juce_image_formats_source "${_juce_image_formats_source}") string(REPLACE " ImageFileFormat* formats[4];" " ImageFileFormat* formats[3];" _juce_image_formats_source "${_juce_image_formats_source}") file(WRITE "${_juce_image_formats_cpp}" "${_juce_image_formats_source}") message(STATUS "YouGlitch diagnostic: JUCE automatic PNG format selection disabled") juce_add_plugin(YouGlitchSMP1024 COMPANY_NAME "YouGlitch Audio" BUNDLE_ID "audio.youglitch.smp1024" PLUGIN_MANUFACTURER_CODE Ygch PLUGIN_CODE Yg24 FORMATS VST3 Standalone PRODUCT_NAME "YouGlitch SMP 1024" DESCRIPTION "1024-step glitch groovebox" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE) juce_generate_juce_header(YouGlitchSMP1024) target_sources(YouGlitchSMP1024 PRIVATE Source/PluginProcessor.cpp Source/PluginProcessor.h Source/PluginEditor.cpp Source/PluginEditor.h Source/StreamingSampleReader.cpp Source/StreamingSampleReader.h Source/StreamingVoicePool.cpp Source/StreamingVoicePool.h) target_compile_features(YouGlitchSMP1024 PRIVATE cxx_std_17) target_compile_definitions(YouGlitchSMP1024 PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 YOUGLITCH_DISABLE_STREAMING=1) target_link_libraries(YouGlitchSMP1024 PRIVATE juce::juce_audio_utils juce::juce_audio_processors juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags)
